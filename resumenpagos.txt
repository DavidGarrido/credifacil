● 📘 SISTEMA DE PERIODOS DE LIBERCOL - DOCUMENTACIÓN COMPLETA

  🎯 Concepto Fundamental: Periodos vs Cuotas

  En Libercol, NO se pagan "cuotas" individuales. Se pagan "PERIODOS".

  Diferencia Clave:

  - Cuota: Fecha de vencimiento específica (ej: 15 de enero)
  - Periodo: Rango de fechas definido por el día de corte (ej: 6 de diciembre - 5 de enero)

  ---
  🗓️ 1. Lógica de Periodos (getPeriodForDate)

  Ubicación: Installments.vue:1216-1260

  Regla Principal:

  Un periodo es un rango de fechas que comienza el día siguiente al día de corte y termina en el día de corte del siguiente mes.

  Ejemplo con día de corte = 5:

  | Fecha de Cuota | Pertenece al Periodo              |
  |----------------|-----------------------------------|
  | 3 de enero     | 6 dic - 5 ene (periodo anterior)  |
  | 5 de enero     | 6 dic - 5 ene (periodo anterior)  |
  | 6 de enero     | 6 ene - 5 feb (periodo actual)    |
  | 20 de enero    | 6 ene - 5 feb (periodo actual)    |
  | 5 de febrero   | 6 ene - 5 feb (periodo actual)    |
  | 6 de febrero   | 6 feb - 5 mar (periodo siguiente) |

  getPeriodForDate(date, cutoffDay) {
      const year = date.getFullYear();
      const month = date.getMonth();
      const day = date.getDate();

      let startDate, endDate;

      // Regla 1: Si el día <= día de corte, pertenece al periodo que TERMINA este mes
      if (day <= cutoffDay) {
          // Periodo: (corte mes anterior + 1) → (corte mes actual)
          const prevMonth = month === 0 ? 11 : month - 1;
          const prevYear = month === 0 ? year - 1 : year;

          // Ajustar si el mes anterior tiene menos días
          const lastDayPrevMonth = new Date(prevYear, prevMonth + 1, 0).getDate();
          const actualCutoffDayPrev = Math.min(cutoffDay, lastDayPrevMonth);

          startDate = new Date(prevYear, prevMonth, actualCutoffDayPrev + 1);

          // Ajustar si el mes actual tiene menos días
          const lastDayCurrentMonth = new Date(year, month + 1, 0).getDate();
          const actualCutoffDay = Math.min(cutoffDay, lastDayCurrentMonth);

          endDate = new Date(year, month, actualCutoffDay);
      }
      // Regla 2: Si el día > día de corte, pertenece al periodo que COMIENZA este mes
      else {
          // Periodo: (corte mes actual + 1) → (corte mes siguiente)
          startDate = new Date(year, month, cutoffDay + 1);

          const nextMonth = month === 11 ? 0 : month + 1;
          const nextYear = month === 11 ? year + 1 : year;

          const lastDayNextMonth = new Date(nextYear, nextMonth + 1, 0).getDate();
          const actualCutoffDayNext = Math.min(cutoffDay, lastDayNextMonth);

          endDate = new Date(nextYear, nextMonth, actualCutoffDayNext);
      }

      return {
          period_key: `${startDate.getFullYear()}-${startDate.getMonth()}-${startDate.getDate()}_${endDate.getFullYear()}-${endDate.getMonth()}-${endDate.getDate()}`,
          start_date: startDate,
          end_date: endDate
      };
  }

  📊 2. Agrupación de Cuotas en Periodos (generatePeriodPayments)

  Ubicación: Installments.vue:1014-1113

  Proceso:

  1. Agrupa cuotas por el periodo al que pertenecen (según getPeriodForDate)
  2. Suma capital, intereses, seguro y pagos realizados
  3. Calcula estado del periodo (pendiente/parcial/pagado/vencido)

  generatePeriodPayments() {
      const periodGroups = {};
      const cutoffDay = parseInt(this.credit.cutoff_day);

      // Paso 1: Agrupar cuotas por periodo
      this.installments.forEach(installment => {
          if (!installment.due_date) return;

          const dueDate = new Date(installment.due_date);
          const periodInfo = this.getPeriodForDate(dueDate, cutoffDay);
          const key = periodInfo.period_key;

          if (!periodGroups[key]) {
              periodGroups[key] = {
                  period_key: key,
                  start_date: periodInfo.start_date,
                  end_date: periodInfo.end_date,
                  installments: [],
                  total_amount: 0,
                  principal_amount: 0,
                  interest_amount: 0,
                  insurance_amount: 0,
                  paid_amount: 0,
                  status: 'pendiente'
              };
          }

          periodGroups[key].installments.push(installment);

          // Paso 2: Sumar montos
          periodGroups[key].total_amount += parseFloat(installment.total_amount || 0);
          periodGroups[key].principal_amount += parseFloat(installment.principal_amount || 0);
          periodGroups[key].interest_amount += parseFloat(installment.interest_amount || 0);
          periodGroups[key].insurance_amount += parseFloat(installment.insurance_amount || 0);
          periodGroups[key].paid_amount += parseFloat(installment.paid_amount || 0);
      });

      // Paso 3: Calcular estado de cada periodo
      Object.keys(periodGroups).forEach(key => {
          const period = periodGroups[key];

          // Verificar pago anticipado
          if (this.isPaidBeforePeriodStart(period) &&
              !period.installments.some(i => i.status === 'parcial')) {
              const allPaid = period.installments.every(i =>
                  i.status === 'pagado' || i.status === 'pagada'
              );
              if (allPaid) {
                  period.status = 'pagado';
                  return;
              }
          }

          // Verificar pago completo
          const isPaidExactly = Math.abs(period.paid_amount - period.total_amount) < 0.01;
          const isPartiallyPaid = period.paid_amount > 0 && period.paid_amount < period.total_amount;

          if (isPaidExactly) {
              period.status = 'pagado';
          } else if (isPartiallyPaid) {
              period.status = 'parcial';
          }
      });

      // Convertir a array y ordenar
      this.periodPayments = Object.values(periodGroups).sort((a, b) => {
          return new Date(a.start_date) - new Date(b.start_date);
      });
  }

  💰 3. Pagos Anticipados (isPaidBeforePeriodStart)

  Ubicación: Installments.vue:1429-1450

  Concepto:

  Un pago anticipado ocurre cuando pagas un periodo ANTES de que comience su fecha de inicio.

  Consecuencia:

  - NO se cobran intereses (solo capital + seguro)
  - Se marca como "Pagado (anticipado)" en la UI

  isPaidBeforePeriodStart(periodSummary) {
      if (!periodSummary || !periodSummary.installments ||
          periodSummary.installments.length === 0 || periodSummary.paid_amount <= 0) {
          return false;
      }

      const startDate = new Date(periodSummary.start_date);

      // Verificar si alguna cuota tiene payment_date anterior al inicio del periodo
      return periodSummary.installments.some(installment => {
          if (!installment.payment_date || installment.status === 'pendiente') {
              return false;
          }

          const paymentDate = new Date(installment.payment_date);
          return paymentDate < startDate;
      });
  }

  Ejemplo:

  - Periodo: 6 enero - 5 febrero
  - Pago realizado: 20 de diciembre
  - Resultado: Pago anticipado → NO cobra intereses del periodo

  🔄 6. Orden de Aplicación de Pagos

  Pago Normal (dentro del periodo o después):

  1. Interés
  2. Seguro
  3. Capital

  Pago Anticipado (antes del periodo):

  1. Seguro
  2. Capital
  3. ❌ NO interés

  🎯 Resumen Ejecutivo

  El sistema de periodos de Libercol funciona así:

  1. Periodos se definen por día de corte, NO por mes calendario
  2. Una compra puede generar múltiples cuotas en el MISMO periodo
  3. Los pagos se agrupan por periodo, no por cuota individual
  4. Pagos anticipados NO cobran interés, solo capital + seguro
  5. El orden de aplicación cambia según si el pago es anticipado o no

  Diferencia clave vs sistema tradicional:
  - ❌ Sistema tradicional: Pagar cuota 1 → cuota 2 → cuota 3
  - ✅ Sistema Libercol: Pagar periodo 1 (con 3 cuotas adentro) → periodo 2 → etc

  /**
   * Obtiene resumen de cuotas agrupadas por periodo de pago
   * 
   * @param int $landlordCreditId
   * @return array
   */
  public function getPeriodSummary(int $landlordCreditId): array
  {
      // Obtener configuración del crédito
      $installmentParent = CreditInstallment::where('landlord_credit_id', $landlordCreditId)
          ->where('installment_number', 0)
          ->first();

      if (!$installmentParent) {
          throw new \Exception("No se encontró el crédito para landlord_credit_id: $landlordCreditId");
      }

      $creditConfig = $installmentParent->metadata['credit_config'] ?? [];
      $cutoffDay = $creditConfig['cutoff_day'] ?? 1;

      // Obtener TODAS las cuotas de TODAS las compras de este crédito
      $allInstallments = CreditInstallment::where('landlord_credit_id', $landlordCreditId)
          ->where('installment_number', '>', 0) // Excluir el parent (número 0)
          ->orderBy('due_date')
          ->get();

      // Agrupar por periodo
      $periodGroups = [];

      foreach ($allInstallments as $installment) {
          $periodInfo = $this->getPeriodForDate($installment->due_date, $cutoffDay);
          $periodKey = $periodInfo['period_key'];

          if (!isset($periodGroups[$periodKey])) {
              $periodGroups[$periodKey] = [
                  'period_key' => $periodKey,
                  'start_date' => $periodInfo['start_date'],
                  'end_date' => $periodInfo['end_date'],
                  'installments' => [],
                  'total_amount' => 0,
                  'principal_amount' => 0,
                  'interest_amount' => 0,
                  'insurance_amount' => 0,
                  'paid_amount' => 0,
              ];
          }

          $periodGroups[$periodKey]['installments'][] = $installment;
          $periodGroups[$periodKey]['total_amount'] += $installment->total_amount;
          $periodGroups[$periodKey]['principal_amount'] += $installment->principal_amount;
          $periodGroups[$periodKey]['interest_amount'] += $installment->interest_amount;
          $periodGroups[$periodKey]['insurance_amount'] += $installment->insurance_amount;
          $periodGroups[$periodKey]['paid_amount'] += $installment->paid_amount;
      }

      // Calcular estado de cada periodo
      foreach ($periodGroups as $key => $period) {
          $allPaid = collect($period['installments'])->every(fn($i) => $i->status === 'pagada');
          $somePaid = collect($period['installments'])->some(fn($i) => $i->paid_amount > 0);

          if ($allPaid) {
              $periodGroups[$key]['status'] = 'pagado';
          } elseif ($somePaid) {
              $periodGroups[$key]['status'] = 'parcial';
          } else {
              $periodGroups[$key]['status'] = 'pendiente';
          }
      }

      return array_values($periodGroups);
  }

  3. Método getPeriodForDate (NUEVO)
  /**
   * Determina el periodo al que pertenece una fecha según el día de corte
   * 
   * @param string $dueDate
   * @param int $cutoffDay
   * @return array
   */
  private function getPeriodForDate(string $dueDate, int $cutoffDay): array
  {
      $date = Carbon::parse($dueDate);
      $year = $date->year;
      $month = $date->month;
      $day = $date->day;

      // Si el día <= día de corte, pertenece al periodo que TERMINA este mes
      if ($day <= $cutoffDay) {
          // Periodo: (corte mes anterior + 1) → (corte mes actual)
          $startDate = $date->copy()->subMonth()->day(min($cutoffDay, $date->copy()->subMonth()->daysInMonth))->addDay();
          $endDate = $date->copy()->day(min($cutoffDay, $date->daysInMonth));
      }
      // Si el día > día de corte, pertenece al periodo que COMIENZA este mes
      else {
          // Periodo: (corte mes actual + 1) → (corte mes siguiente)
          $startDate = $date->copy()->day($cutoffDay)->addDay();
          $endDate = $date->copy()->addMonth()->day(min($cutoffDay, $date->copy()->addMonth()->daysInMonth));
      }

      return [
